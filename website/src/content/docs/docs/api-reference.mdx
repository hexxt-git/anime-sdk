## `createSdk(opts?)`

```ts
import { createSdk } from 'anime-sdk';
const sdk = createSdk(opts?);
```

Factory — instantiates the SDK with all 12 sources registered.

### `SdkOptions`

```ts
interface SdkOptions {
  sources?: string[]; // whitelist source IDs; default: all 12
  disabled?: string[]; // blacklist a subset
  http?: {
    timeoutMs?: number; // default: 30000
    retries?: number; // default: 3
    userAgent?: string;
  };
}
```

## Sdk methods

### `sdk.search(query, opts?)`

```ts
sdk.search(query: string, opts?: {
  kind?:   'anime' | 'manga';  // default: 'anime'
  signal?: AbortSignal;
}): ProgressiveResult<Media>
```

Returns a value that is both `AsyncIterable<Media>` (iterate for results as they arrive) and `PromiseLike<Media[]>` (await for the full array). Calls `.cancel()` to abort.

### `sdk.info(media, opts?)`

```ts
sdk.info(media: Media | string, opts?: { signal?: AbortSignal }): Promise<Media>
```

Full info for a single title. Accepts a `Media` object or an opaque `id` string.

### `sdk.sources(media, opts?)`

```ts
sdk.sources(media: Media | string, opts?: { signal?: AbortSignal }): Promise<SourceInfo[]>
```

Returns playback sources ranked by health × coverage. `status: 'available'` means the source has this title.

### `sdk.episodes(media, opts?)`

```ts
sdk.episodes(media: Media | string, opts?: {
  signal?:  AbortSignal;
  cursor?:  string;
  limit?:   number;
}): Promise<List<Episode>>
```

### `sdk.chapters(media, opts?)`

```ts
sdk.chapters(media: Media | string, opts?: {
  signal?:  AbortSignal;
  cursor?:  string;
  limit?:   number;
}): Promise<List<Chapter>>
```

### `sdk.stream(episode, opts?)`

```ts
sdk.stream(episode: Episode | string, opts?: {
  signal?: AbortSignal;
}): ProgressiveResult<Stream>
```

Returns **all streams** available for the episode — all sources, all servers, all languages — as a `ProgressiveResult<Stream>` (async-iterable or `await`able for the full array). Each `Stream` is one playable URL with `source`, `server`, `quality`, and `language` metadata; the frontend picks the preferred one.

With an `Episode` object, the SDK auto-fans-out across all enabled playback sources (using internally cached media from the prior `episodes()` call). With a string ID, only the source encoded in the ID is queried.

### `sdk.pages(chapter, opts?)`

```ts
sdk.pages(chapter: Chapter | string, opts?: { signal?: AbortSignal }): Promise<Pages>
```

No `language` argument — manga has no language axis.

### `sdk.browse(opts)`

```ts
sdk.browse(opts: {
  list:     'trending' | 'popular' | 'seasonal' | 'top';
  kind:     'anime' | 'manga';
  signal?:  AbortSignal;
  page?:    number;
  perPage?: number;
  season?:  string;  // required for 'seasonal'
  year?:    number;  // required for 'seasonal'
}): Promise<List<Media>>
```

### `sdk.health()`

```ts
sdk.health(): SourceHealth[]
```

Synchronous. Returns rolling success/latency stats per source.

## Value types

### `Media`

```ts
interface Media {
  id: string; // opaque base64url token — pass to SDK methods
  kind: 'anime' | 'manga';
  title: MediaTitle;
  cover?: MediaCover;
  banner?: string;
  score?: Score;
  year?: number;
  season?: 'WINTER' | 'SPRING' | 'SUMMER' | 'FALL';
  status?: 'FINISHED' | 'RELEASING' | 'NOT_YET_RELEASED' | 'CANCELLED' | 'HIATUS';
  format?: 'TV' | 'MOVIE' | 'OVA' | 'ONA' | 'SPECIAL' | 'MANGA' | 'NOVEL';
  episodeCount?: number;
  chapterCount?: number;
  description?: string;
  source: string; // which source produced this record
  mappings: {
    anilist?: number;
    mal?: number;
    kitsu?: number;
  };
}
```

### `MediaTitle`

```ts
interface MediaTitle {
  preferred: string; // best available: english ?? romaji ?? native
  english?: string;
  romaji?: string;
  native?: string;
}
```

### `MediaCover`

```ts
interface MediaCover {
  url: string;
  color?: string; // dominant color in hex (e.g. '#e4a15d')
}
```

### `Score`

```ts
interface Score {
  value: number; // e.g. 87
  scale: number; // e.g. 100
}
// Display as: (score.value / score.scale * 10).toFixed(1) → "8.7"
```

### `Episode`

```ts
interface Episode {
  id: string; // opaque — pass to sdk.stream()
  number: number;
  title?: string;
  thumbnail?: string;
  airDate?: string;
  filler?: boolean;
  recap?: boolean;
  languages?: ('sub' | 'dub' | 'raw')[];
}
```

### `Chapter`

```ts
interface Chapter {
  id: string; // opaque — pass to sdk.pages()
  number: number;
  title?: string;
}
```

### `Stream`

Each `Stream` is one playable URL. Sources that offer multiple servers or qualities return multiple `Stream` objects.

```ts
interface Stream {
  url: string;
  source: string; // which source provided this stream
  server: string; // server within the source (e.g. 'mp4upload', 'wixmp')
  quality: '1080p' | '720p' | '480p' | '360p' | 'auto';
  language: 'sub' | 'dub' | 'raw';
  isHls: boolean;
  headers?: Record<string, string>;
  subtitles: Subtitle[];
}
```

### `Pages`

```ts
interface Pages {
  pages: { url: string; width?: number; height?: number }[];
}
```

### `Subtitle`

```ts
interface Subtitle {
  url: string;
  language: string; // BCP-47 (e.g. 'en', 'pt-BR')
  label: string; // human-readable ('English')
  format: 'vtt' | 'srt' | 'ass';
}
```

### `List<T>`

```ts
interface List<T> {
  items: T[];
  nextCursor?: string; // undefined when exhausted
  total?: number;
}
```

### `SourceInfo`

```ts
interface SourceInfo {
  id: string;
  status: 'available' | 'incompatible' | 'error';
  episodeCount?: number;
  successRate?: number; // rolling 0–1 from last 20 calls
}
```

### `ProgressiveResult<T>`

```ts
interface ProgressiveResult<T> extends AsyncIterable<T>, PromiseLike<T[]> {
  cancel(): void;
}
```

## Errors

### `AniError`

```ts
class AniError extends Error {
  readonly code: AniErrorCode;
  readonly source?: string; // source id, when known
  readonly retryable: boolean;
  readonly cause?: unknown;
}
```

### `AniErrorCode`

```ts
const AniErrorCode = {
  SourceUnavailable: 'SourceUnavailable', // upstream unreachable
  NoStream: 'NoStream', // no playable URL found
  RegionBlocked: 'RegionBlocked', // geo-restricted content
  RateLimited: 'RateLimited', // rate limit hit
  NotFound: 'NotFound', // title/episode not found
  Cancelled: 'Cancelled', // AbortSignal fired
  BadId: 'BadId', // malformed opaque id
} as const;
```

## Server

### `startServer(opts?)`

```ts
import { startServer } from 'anime-sdk';

function startServer(opts?: {
  port?: number; // default: 0 (random); set PORT env var for 3030
  sdk?: Sdk; // default: createSdk()
  proxy?: ProxyOptions; // enable /proxy + URL rewriting
}): http.Server;
```

Returns an `http.Server`. `server.close()` to shut down. See [HTTP Server](/docs/http-server/) for all routes and [Stream Proxy](/docs/proxy/) for proxy configuration.

### `ProxyOptions`

```ts
interface ProxyOptions {
  base?: string; // public base URL override (defaults to Host header)
  signSecret?: string; // HMAC-SHA256 sign every proxied URL
  allowedHosts?: string[]; // suffix-matched SSRF allowlist
}
```

## Download utilities

### `downloadVideo(stream, outputPath, opts?)`

```ts
import { downloadVideo } from 'anime-sdk';

function downloadVideo(
  stream: Stream,
  outputPath: string,
  opts?: {
    onProgress?: (info: { phase: string; detail?: string }) => void;
    timeoutMs?: number; // default 300_000 (5 min)
    headers?: Record<string, string>; // override stream.headers
  },
): Promise<{ outputPath: string; fileSize: number }>;
```

HLS streams: walks master → variant, downloads every segment (stripping PNG-disguised TS bytes), then `ffmpeg -c copy` muxes to MP4. Direct MP4 streams to disk. Each `Stream` is one URL — if it fails, the caller should try the next `Stream`.

### `downloadMangaChapter(pages, outputPath, opts?)`

```ts
import { downloadMangaChapter } from 'anime-sdk';

function downloadMangaChapter(
  pages: Pages,
  outputPath: string,
  opts?: {
    onProgress?: (info: { downloaded: number; total: number }) => void;
    timeoutMs?: number; // per page; default 30_000
    headers?: Record<string, string>;
  },
): Promise<{ outputPath: string; pageCount: number; fileSize: number }>;
```

Packages all pages as an uncompressed `.zip`.

### `downloadMangaPage(pages, pageIndex, outputDir, opts?)`

```ts
import { downloadMangaPage } from 'anime-sdk';

function downloadMangaPage(
  pages: Pages,
  pageIndex: number,
  outputDir: string,
  opts?: { headers?: Record<string, string>; timeoutMs?: number },
): Promise<{ outputPath: string; pageIndex: number; fileSize: number; contentType: string }>;
```
